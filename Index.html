require(patentId > 0 && patentId <= patentCounter, "OmniPatent: Invalid ID"); require(msg.sender == p.inventor || msg.sender == hubAdmin, "OmniPatent: Unauthorized"); require(p.isAvailable, "OmniPatent: Already unavailable"); p.isAvailable = false; emit OfferCanceled(patentId); } // 3. depositPlatformTreasury function depositPlatformTreasury() external payable whenNotPaused { require(msg.value > 0, "OmniPatent: Zero deposit"); accumulatedFees += msg.value; emit TreasuryDeposited(msg.sender, msg.value); } // 4. drainHubReserves function drainHubReserves() external onlyAdmin { uint256 balance = address(this).balance; require(balance > 0, "OmniPatent: Zero balance"); accumulatedFees = 0; (bool sent, ) = hubAdmin.call{value: balance}(""); require(sent, "OmniPatent: Sweep failed"); emit HubReservesDrained(hubAdmin, balance); } // 5. harvestPlatformFees function harvestPlatformFees() external onlyAdmin { uint256 amount = accumulatedFees; require(amount > 0, "OmniPatent: Zero fees"); accumulatedFees = 0; (bool sent, ) = hubAdmin.call{value: amount}(""); require(sent, "OmniPatent: Harvest failed"); emit FeesHarvested(hubAdmin, amount); } // 6. pauseLicensingHub function pauseLicensingHub(bool isPaused) external onlyAdmin { isHubPaused = isPaused; emit HubPauseToggled(isPaused); } // 7. registerNewPatent function registerNewPatent(uint256 licenseFeeWei, bytes32 documentHash) external whenNotPaused returns (uint256) { require(licenseFeeWei > 0, "OmniPatent: Free licenses not supported via hub"); patentCounter++; patents[patentCounter] = Patent({ inventor: payable(msg.sender), licenseFeeWei: licenseFeeWei, isAvailable: true, documentHash: documentHash }); emit PatentRegistered(patentCounter, msg.sender, licenseFeeWei); return (p.inventor, p.licenseFeeWei, p.isAvailable, p.documentHash); } }
